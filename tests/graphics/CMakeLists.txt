cmake_minimum_required(VERSION 3.20)
project(FrogN64ProjectionTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(GLM_INCLUDE_DIR glm/glm.hpp REQUIRED)
enable_testing()
add_executable(projection-tests projection_tests.cpp)
target_include_directories(projection-tests PRIVATE ../../Source ${GLM_INCLUDE_DIR})
set(FROGN64_SANITIZER "address,undefined" CACHE STRING "Sanitizers; empty disables")
if(FROGN64_SANITIZER AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(projection-tests PRIVATE -fsanitize=${FROGN64_SANITIZER} -fno-omit-frame-pointer)
    target_link_options(projection-tests PRIVATE -fsanitize=${FROGN64_SANITIZER})
endif()
add_test(NAME cached-vertex-projections COMMAND projection-tests)
set_tests_properties(cached-vertex-projections PROPERTIES TIMEOUT 10)

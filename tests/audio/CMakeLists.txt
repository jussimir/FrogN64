cmake_minimum_required(VERSION 3.20)
project(FrogN64AudioTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
enable_testing()
add_executable(audio-buffer-tests buffer_tests.cpp ../../Source/HLEAudio/AudioBuffer.cpp)
target_include_directories(audio-buffer-tests PRIVATE ../../Source)
target_compile_definitions(audio-buffer-tests PRIVATE DAEDALUS_ENDIAN_LITTLE=1 DAEDALUS_POSIX=1)
target_link_libraries(audio-buffer-tests PRIVATE Threads::Threads)
set(FROGN64_SANITIZER "address,undefined" CACHE STRING "Sanitizers; empty disables")
if(FROGN64_SANITIZER AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(audio-buffer-tests PRIVATE -fsanitize=${FROGN64_SANITIZER} -fno-omit-frame-pointer)
    target_link_options(audio-buffer-tests PRIVATE -fsanitize=${FROGN64_SANITIZER})
endif()
foreach(case fifo silence interpolation packets wrap full)
    add_test(NAME audio-${case} COMMAND audio-buffer-tests ${case})
    set_tests_properties(audio-${case} PROPERTIES TIMEOUT 10)
endforeach()
